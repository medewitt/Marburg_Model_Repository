% Speed-strength plane for Marburg outbreaks, with uncertainty in both axes.
% x = intrinsic growth rate r (day^-1), 95% CI as a horizontal whisker.
% y = mean generation interval T_g (days). T_g is era-specific: pre-2014
%     outbreaks are centred at 9.3 d and 2020s outbreaks at 11 d, each with a
%     95% band from Normal(mu, 1.0) (vertical whisker).
% R0 contours use the gamma generation-interval form (Wallinga & Lipsitch 2007)
%     R0 = (1 + r*T_g/kappa)^kappa at a representative CV = 0.6 (kappa = 1/CV^2),
%     so T_g = kappa*(R0^{1/kappa}-1)/r. Point R0 estimates (median, 95%) come
%     from the Monte Carlo in outbreak_R0_montecarlo.csv, which also samples CV.
% Reading where an outbreak sits relative to the contours gives its R0.
%
% Compile with pdflatex (or lualatex) + pgfplots (>= 1.16).
\documentclass[border=3pt]{standalone}
\usepackage{pgfplots}
\pgfplotsset{compat=1.16}
\usepackage{amsmath}

% Okabe-Ito colorblind-safe palette
\definecolor{oiOrange}{RGB}{230,159,0}
\definecolor{oiGreen}{RGB}{0,158,115}
\definecolor{oiBlue}{RGB}{0,114,178}
\definecolor{oiVermil}{RGB}{213,94,0}
\definecolor{oiBlack}{RGB}{0,0,0}

\begin{document}
\begin{tikzpicture}
\begin{axis}[
    width=13cm, height=9.5cm,
    xlabel={Intrinsic growth rate $r$ (day$^{-1}$)},
    ylabel={Mean generation interval $T_g$ (days)},
    xmin=-0.04, xmax=0.32, ymin=0, ymax=14,
    axis lines=left,
    xtick={0,0.05,0.10,0.15,0.20,0.25,0.30},
    ytick={0,2,4,6,8,9.3,11,14}, yticklabels={0,2,4,6,8,9.3,11,14},
    tick label style={font=\small}, label style={font=\small}, clip=true,
]

% ---- era reference lines (mean T_g) ----
\draw[gray!55,dashed] (axis cs:-0.04,9.3) -- (axis cs:0.32,9.3);
\draw[gray!55,dashed] (axis cs:-0.04,11) -- (axis cs:0.32,11);
\node[gray,font=\scriptsize,anchor=east] at (axis cs:0.318,8.7){pre-2014};
\node[gray,font=\scriptsize,anchor=east] at (axis cs:0.318,11.5){2020s};

% ---- gamma-GI R0 contours at CV=0.6: T_g = c/r ----
\addplot[gray!55,thick,domain=0.0307:0.32,samples=100]{0.430/x};
\addplot[gray!55,thick,domain=0.0562:0.32,samples=100]{0.787/x};
\addplot[gray!55,thick,domain=0.0980:0.32,samples=100]{1.372/x};
\addplot[gray!55,thick,domain=0.1271:0.32,samples=100]{1.780/x};
\node[gray,font=\scriptsize] at (axis cs:0.036,12.9){$R_0{=}1.5$};
\node[gray,font=\scriptsize] at (axis cs:0.064,12.9){$R_0{=}2$};
\node[gray,font=\scriptsize] at (axis cs:0.109,12.9){$R_0{=}3$};
\node[gray,font=\scriptsize] at (axis cs:0.142,12.9){$R_0{=}4$};

% ---- R0 = 1 threshold ----
\draw[gray,dashed] (axis cs:0,0) -- (axis cs:0,14);
\node[gray,font=\scriptsize,rotate=90,anchor=south] at (axis cs:0,2.2){$R_0=1$};
